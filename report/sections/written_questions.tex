\begin{question}[1. Insertion Sort]{
Implement \code{insertion_sort(…)} in \code{sort.c}. The function takes three arguments: an array of integers, low and high indices to denote the range to be sorted. Thus, the function sorts the array of range \code{[low, high)}. The function must produce the array of integers, whose elements are located in ascending order.
}
%\begin{answer}
\begin{lstlisting}[language=C, style=C]
void insertion_sort(int arr[], int low, int high)
{
    for (int i = low + 1; i < high; i++) {
        int key = arr[i];
        int j = i - 1;
        while (j >= low && arr[j] > key) {
            arr[j + 1] = arr[j];
            j--;
        }
        arr[j + 1] = key;
    }
}
\end{lstlisting}

\begin{question_with_subs}[2. Merge Sort]{}
%%%%% A %%%%%
\begin{subq}{Implement \code{merge(…)} in \code{sort.c}. The function takes four arguments: an array of integers, \code{low}, \code{mid}, and \code{high} indices. The \code{low} and \code{high} indices denote the range to be sorted. The \code{mid} is defined by \code{(low + high) / 2}. Therefore, the \code{low} and the \code{mid} denote the start indices of the left and the right subarrays, respectively. The output of the function must be the sorted array of size \code{high - low}.}
\begin{lstlisting}[language=C, style=C]
void merge(int arr[], int low, int mid, int high)
{
    int n1 = mid - low;
    int n2 = high - mid;

    int* left = (int*)malloc(sizeof(int) * n1);
    int* right = (int*)malloc(sizeof(int) * n2);

    for (int i = 0; i < n1; i++)
        left[i] = arr[low + i];
    for (int j = 0; j < n2; j++)
        right[j] = arr[mid + j];

    int i = 0, j = 0, k = low;
    while (i < n1 && j < n2) {
        if (left[i] <= right[j]) {
            arr[k++] = left[i++];
        } else {
            arr[k++] = right[j++];
        }
    }
    while (i < n1)
        arr[k++] = left[i++];
    while (j < n2)
        arr[k++] = right[j++];

    free(left);
    free(right);
}
\end{lstlisting}
\end{subq}

void merge_sort(int arr[], int low, int high)
{
    if (high - low <= 1) return;
    int mid = low + (high - low) / 2;
    merge_sort(arr, low, mid);
    merge_sort(arr, mid, high);
    merge(arr, low, mid, high);
}
\end{lstlisting}
\end{subq}
\end{question_with_subs}

\begin{question_with_subs}[3. Quick Sort]{}
%%%%% A %%%%%
\begin{subq}{Implement \code{partition(…)} in \code{sort.c}. The function takes three arguments: an array of integers, \code{low}, and \code{high} indices. The \code{low} and \code{high} indices denote the range to be partitioned.The function outputs the index of pivot in the partitioned array. For the partitioning, you should select random pivot. Also, you can use \code{swap()} in \code{util.h}}
\begin{lstlisting}[language=C, style=C]
int partition(int arr[], int low, int high)
{
    int pivot = arr[high - 1];
    int i = low - 1;
    for (int j = low; j < high - 1; j++) {
        if (arr[j] <= pivot) {
            i++;
            swap(&arr[i], &arr[j]);
        }
    }
    swap(&arr[i + 1], &arr[high - 1]);
    return i + 1;
}
\end{lstlisting}
\end{subq}

void quick_sort(int arr[], int low, int high)
{
    if (low < high - 1) {
        int p = partition(arr, low, high);
        quick_sort(arr, low, p);
        quick_sort(arr, p + 1, high);
    }
}
\end{lstlisting}
\end{subq}
